      className="bg-white p-8 rounded-3xl border border-slate-100 shadow-sm hover:shadow-md transition-shadow cursor-pointer"

    >
      <div className="text-brand-primary mb-6">{icon}</div>
      <h3 className="text-xl font-bold text-slate-900 mb-3">{title}</h3>
      <p className="text-slate-600 text-sm leading-relaxed mb-4">{desc}</p>
      
      <div className={`overflow-hidden transition-all duration-300 ${isOpen ? 'max-h-40' : 'max-h-0'}`}>
        <p className="pt-4 border-t border-slate-50 text-sm text-slate-500 italic">
          {details}
        </p>
      </div>
      
      <button className="mt-4 text-brand-primary text-xs font-bold uppercase tracking-wider flex items-center gap-1">
        {isOpen ? 'Less' : 'Learn More'} <ChevronRight className={`w-3 h-3 transition-transform ${isOpen ? 'rotate-90' : ''}`} />
      </button>
    </div>
  );
}
